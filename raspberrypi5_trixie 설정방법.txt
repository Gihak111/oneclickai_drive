## 서보관련 hardware pwm 사용하도록 (서보 안쓰면 상관 없음)
#단, os 버전이 너무 낮으면 동작 안할수도 있음. (trixie에서는 확실히 동작)
sudo nano /boot/firmware/config.txt
# 그리고 아래 [all] 밑에 한 줄만 추가:
dtoverlay=pwm,pin=18,func=2

# 파일 추가 service
# pwm18.service 파일을 /etc/systemd/system로 이동
# pwm18_prepare.sh 파일을 /home/pi/settings로 이동 (폴더 새로 만들어야함)

# 터미널 창에
systemctl enable --now pwm18.service



## 웹서비스 등록 (웹 구동시 필요. 웹 구동 안하면 안해도 됨. 사용한다면 아이피 꼭 위에 설정대로 192.168.137.18로 고정해야함)

# caddy 설치
sudo apt install -y caddy
sudo install -d -o root -g caddy -m 750 /etc/caddy/certs/auto.oneclickai.work

# caddy 권한 설정
sudo usermod -aG caddy pi
# 재부팅
 
# Caddyfile /etc/caddy 폴더로 이동
# 서비스 파일, /etc/systemd/system로 이동
# 서비스 시작
systemctl enable --now autocert.service

# 웹 자동화
# 파일이동 autoweb.desktop 파일은 여기로 이동
/home/pi/.config/autostart/
